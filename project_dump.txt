class Car {
  final String id;
  final String userId;
  final String brand;
  final String model;
  final int year;
  final String plate;
  final int mileage;
  final DateTime createdAt;

  Car({
    required this.id,
    required this.userId,
    required this.brand,
    required this.model,
    required this.year,
    required this.plate,
    required this.mileage,
    required this.createdAt,
  });

  String get title => '$brand $model';

  factory Car.fromMap(Map<String, dynamic> map, String id) {
    return Car(
      id: id,
      userId: map['userId'] ?? '',
      brand: map['brand'] ?? '',
      model: map['model'] ?? '',
      year: map['year'] ?? 0,
      plate: map['plate'] ?? '',
      mileage: map['mileage'] ?? 0,
      createdAt: (map['createdAt'] as Timestamp?)?.toDate() ?? DateTime.now(),
    );
  }

  Map<String, dynamic> toMap() {
    return {
      'userId': userId,
      'brand': brand,
      'model': model,
      'year': year,
      'plate': plate,
      'mileage': mileage,
      'createdAt': Timestamp.fromDate(createdAt),
    };
  }
}

enum ExpenseType { fuel, service, tires, insurance, tax, wash, other }

extension ExpenseTypeX on ExpenseType {
  String get label {
    switch (this) {
      case ExpenseType.fuel:
        return 'Топливо';
      case ExpenseType.service:
        return 'ТО / Ремонт';
      case ExpenseType.tires:
        return 'Шины';
      case ExpenseType.insurance:
        return 'Страховка';
      case ExpenseType.tax:
        return 'Налог';
      case ExpenseType.wash:
        return 'Мойка';
      case ExpenseType.other:
        return 'Прочее';
    }
  }

  String get icon {
    switch (this) {
      case ExpenseType.fuel:
        return '⛽';
      case ExpenseType.service:
        return '🔧';
      case ExpenseType.tires:
        return '🛞';
      case ExpenseType.insurance:
        return '📄';
      case ExpenseType.tax:
        return '💰';
      case ExpenseType.wash:
        return '🚿';
      case ExpenseType.other:
        return '📦';
    }
  }
}

class Expense {
  final String id;
  final String carId;
  final ExpenseType type;
  final double amount;
  final int mileage;
  final DateTime date;
  final String note;
  final double? liters;
  final double? pricePerLiter;
  final String? serviceName;
  final String? receiptImagePath;

  Expense({
    required this.id,
    required this.carId,
    required this.type,
    required this.amount,
    required this.mileage,
    required this.date,
    required this.note,
    this.liters,
    this.pricePerLiter,
    this.serviceName,
    this.receiptImagePath,
  });

  factory Expense.fromMap(Map<String, dynamic> map, String id) {
    return Expense(
      id: id,
      carId: map['carId'] ?? '',
      type: ExpenseType.values.firstWhere(
        (e) => e.name == map['type'],
        orElse: () => ExpenseType.other,
      ),
      amount: (map['amount'] ?? 0).toDouble(),
      mileage: map['mileage'] ?? 0,
      date: (map['date'] as Timestamp?)?.toDate() ?? DateTime.now(),
      note: map['note'] ?? '',
      liters: (map['liters'] as num?)?.toDouble(),
      pricePerLiter: (map['pricePerLiter'] as num?)?.toDouble(),
      serviceName: map['serviceName'] as String?,
      receiptImagePath: map['receiptImagePath'] as String?,
    );
  }

  Map<String, dynamic> toMap() {
    return {
      'carId': carId,
      'type': type.name,
      'amount': amount,
      'mileage': mileage,
      'date': Timestamp.fromDate(date),
      'note': note,
      'liters': liters,
      'pricePerLiter': pricePerLiter,
      'serviceName': serviceName,
      'receiptImagePath': receiptImagePath,
    };
  }
}

class AuthProvider extends ChangeNotifier {
  final AuthService _auth = AuthService();
  User? _user;
  bool _loading = false;

  User? get user => _user;
  bool get loading => _loading;
  bool get isSignedIn => _user != null;

  AuthProvider() {
    _user = _auth.currentUser;
    _auth.authStateChanges.listen((user) {
      _user = user;
      notifyListeners();
    });
  }

  Future<bool> signInWithGoogle() async {
    _loading = true;
    notifyListeners();
    try {
      final user = await _auth.signInWithGoogle();
      _user = user;
      return user != null;
    } catch (e) {
      debugPrint('Ошибка входа: $e');
      return false;
    } finally {
      _loading = false;
      notifyListeners();
    }
  }

  Future<void> signOut() async {
    await _auth.signOut();
    _user = null;
    notifyListeners();
  }
}

class DataProvider extends ChangeNotifier {
  final DbService _db = DbService();
  final String userId;

  DataProvider(this.userId) {
    _init();
  }

  List<Car> _cars = [];
  bool _carsLoading = true;
  Map<String, List<Expense>> _expenses = {};

  List<Car> get cars => _cars;
  bool get carsLoading => _carsLoading;

  List<Expense> expensesFor(String carId) => _expenses[carId] ?? [];

  void _init() {
    _db.watchCars(userId).listen((cars) {
      _cars = cars;
      _carsLoading = false;
      notifyListeners();
    });
  }

  Future<void> addCar(Car car) async {
    await _db.addCar(car);
  }

  Future<void> updateCar(Car car) async {
    await _db.updateCar(car);
  }

  Future<void> deleteCar(String id) async {
    await _db.deleteCar(id);
  }

  void subscribeToExpenses(String carId) {
    _db.watchExpenses(carId).listen((list) {
      _expenses[carId] = list;
      notifyListeners();
    });
  }

  Future<void> addExpense(Expense expense) async {
    await _db.addExpense(expense);
    notifyListeners(); // Гарантируем мгновенное обновление UI
  }

  Future<void> updateExpense(Expense expense) async {
    await _db.updateExpense(expense);
    notifyListeners(); // Гарантируем мгновенное обновление UI
  }

  Future<void> deleteExpense(String id) async {
    await _db.deleteExpense(id);
    notifyListeners(); // Гарантируем мгновенное обновление UI
  }
}

class AddCarScreen extends StatefulWidget {
  final Car? car;
  const AddCarScreen({super.key, this.car});

  @override
  State<AddCarScreen> createState() => _AddCarScreenState();
}

class _AddCarScreenState extends State<AddCarScreen> {
  final _formKey = GlobalKey<FormState>();
  final _brandCtrl = TextEditingController();
  final _modelCtrl = TextEditingController();
  final _yearCtrl = TextEditingController();
  final _plateCtrl = TextEditingController();
  final _mileageCtrl = TextEditingController();

  @override
  void initState() {
    super.initState();
    if (widget.car != null) {
      _brandCtrl.text = widget.car!.brand;
      _modelCtrl.text = widget.car!.model;
      _yearCtrl.text = widget.car!.year.toString();
      _plateCtrl.text = widget.car!.plate;
      _mileageCtrl.text = widget.car!.mileage.toString();
    }
  }

  @override
  void dispose() {
    _brandCtrl.dispose();
    _modelCtrl.dispose();
    _yearCtrl.dispose();
    _plateCtrl.dispose();
    _mileageCtrl.dispose();
    super.dispose();
  }

  Future<void> _save() async {
    if (!_formKey.currentState!.validate()) return;

    final car = Car(
      id: widget.car?.id ?? const Uuid().v4(),
      userId: context.read<AuthProvider>().user!.uid,
      brand: _brandCtrl.text.trim(),
      model: _modelCtrl.text.trim(),
      year: int.tryParse(_yearCtrl.text) ?? DateTime.now().year,
      plate: _plateCtrl.text.trim(),
      mileage: int.tryParse(_mileageCtrl.text) ?? 0,
      createdAt: widget.car?.createdAt ?? DateTime.now(),
    );

    if (widget.car != null) {
      await context.read<DataProvider>().updateCar(car);
    } else {
      await context.read<DataProvider>().addCar(car);
    }
    if (mounted) Navigator.pop(context);
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text(
          widget.car != null ? 'Редактировать авто' : 'Новый автомобиль',
        ),
      ),
      body: Form(
        key: _formKey,
        child: ListView(
          padding: const EdgeInsets.all(16),
          children: [
            _field('Марка', _brandCtrl),
            const SizedBox(height: 12),
            _field('Модель', _modelCtrl),
            const SizedBox(height: 12),
            _field('Год выпуска', _yearCtrl, keyboard: TextInputType.number),
            const SizedBox(height: 12),
            _field('Гос. номер', _plateCtrl),
            const SizedBox(height: 12),
            _field('Пробег (км)', _mileageCtrl, keyboard: TextInputType.number),
            const SizedBox(height: 24),
            ElevatedButton(onPressed: _save, child: const Text('Сохранить')),
          ],
        ),
      ),
    );
  }

  Widget _field(
    String label,
    TextEditingController ctrl, {
    TextInputType keyboard = TextInputType.text,
  }) {
    return TextFormField(
      controller: ctrl,
      keyboardType: keyboard,
      decoration: InputDecoration(labelText: label),
      validator: (v) =>
          (v == null || v.trim().isEmpty) ? 'Обязательное поле' : null,
    );
  }
}

class AddExpenseScreen extends StatefulWidget {
  final String carId;
  final Expense? expenseToEdit;
  const AddExpenseScreen({super.key, required this.carId, this.expenseToEdit});

  @override
  State<AddExpenseScreen> createState() => _AddExpenseScreenState();
}

class _AddExpenseScreenState extends State<AddExpenseScreen> {
  final _formKey = GlobalKey<FormState>();
  final OcrService _ocrService = OcrService();
  final ImagePicker _picker = ImagePicker();

  ExpenseType _type = ExpenseType.fuel;
  final _amountCtrl = TextEditingController();
  final _mileageCtrl = TextEditingController();
  final _noteCtrl = TextEditingController();
  final _litersCtrl = TextEditingController();
  final _pricePerLiterCtrl = TextEditingController();
  DateTime _date = DateTime.now();

  String? _selectedServiceId;
  bool _isCustomService = false;
  final _customServiceCtrl = TextEditingController();

  bool _isScanning = false;
  String? _receiptImagePath;

  @override
  void initState() {
    super.initState();
  }

  void _calculatePrice() {
    final amountText = _amountCtrl.text.trim().replaceAll(',', '.');
    final litersText = _litersCtrl.text.trim().replaceAll(',', '.');

    if (amountText.isEmpty || litersText.isEmpty) {
      ScaffoldMessenger.of(context)
          .showSnackBar(const SnackBar(content: Text('Заполните оба поля')));
      return;
    }

    final amount = double.tryParse(amountText);
    final liters = double.tryParse(litersText);

    if (amount != null && liters != null && liters > 0) {
      setState(() {
        _pricePerLiterCtrl.text = (amount / liters).toStringAsFixed(2);
      });
    }
  }

  @override
  void dispose() {
    _amountCtrl.dispose();
    _mileageCtrl.dispose();
    _noteCtrl.dispose();
    _litersCtrl.dispose();
    _pricePerLiterCtrl.dispose();
    _customServiceCtrl.dispose();
    _ocrService.dispose();
    super.dispose();
  }

  Future<void> _save() async {
    if (!_formKey.currentState!.validate()) return;

    double amount = double.parse(_amountCtrl.text.replaceAll(',', '.'));
    double? liters = _litersCtrl.text.isNotEmpty
        ? double.parse(_litersCtrl.text.replaceAll(',', '.'))
        : null;
    double? pricePerLiter = _pricePerLiterCtrl.text.isNotEmpty
        ? double.parse(_pricePerLiterCtrl.text.replaceAll(',', '.'))
        : null;

    if (_type == ExpenseType.fuel &&
        liters != null &&
        liters > 0 &&
        pricePerLiter == null) {
      pricePerLiter = amount / liters;
    } else if (_type == ExpenseType.fuel &&
        liters != null &&
        pricePerLiter != null &&
        _amountCtrl.text.isEmpty) {
      amount = liters * pricePerLiter;
    }

    final expense = Expense(
      id: widget.expenseToEdit?.id ?? const Uuid().v4(),
      carId: widget.carId,
      type: _type,
      amount: amount,
      mileage: int.parse(_mileageCtrl.text),
      date: _date,
      note: _noteCtrl.text.trim(),
      liters: liters,
      pricePerLiter: pricePerLiter,
      serviceName: _isCustomService
          ? _customServiceCtrl.text.trim()
          : _selectedServiceId,
      receiptImagePath: _receiptImagePath,
    );

    if (widget.expenseToEdit != null) {
      await context.read<DataProvider>().updateExpense(expense);
    } else {
      await context.read<DataProvider>().addExpense(expense);
    }

    if (mounted) context.pop();
  }

  Future<void> _pickDate() async {
    final picked = await showDatePicker(
      context: context,
      initialDate: _date,
      firstDate: DateTime(2000),
      lastDate: DateTime.now(),
      locale: const Locale('ru', 'RU'),
    );
    if (picked != null) setState(() => _date = picked);
  }

  Future<void> _scanReceipt() async {
    setState(() => _isScanning = true);
    try {
      final result = await _ocrService.scanReceipt();
      if (result != null) {
        if (result['amount'] != null)
          _amountCtrl.text = result['amount'].toString();
        if (result['liters'] != null)
          _litersCtrl.text = result['liters'].toString();
        if (result['date'] != null) setState(() => _date = result['date']);
      }
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context)
            .showSnackBar(SnackBar(content: Text('Ошибка сканирования: $e')));
      }
    } finally {
      if (mounted) setState(() => _isScanning = false);
    }
  }

  Future<void> _pickImage() async {
    final XFile? image = await _picker.pickImage(source: ImageSource.gallery);
    if (image != null) {
      setState(() => _receiptImagePath = image.path);
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text(
          widget.expenseToEdit != null
              ? 'Редактировать расход'
              : 'Новый расход',
        ),
      ),
      body: Form(
        key: _formKey,
        child: ListView(
          padding: const EdgeInsets.all(16),
          children: [
            Wrap(
              spacing: 8,
              runSpacing: 8,
              children: ExpenseType.values.map((t) {
                final selected = t == _type;
                return ChoiceChip(
                  label: Text('${t.icon} ${t.label}'),
                  selected: selected,
                  onSelected: (_) => setState(() => _type = t),
                  selectedColor: AppTheme.primary.withValues(alpha: 0.2),
                );
              }).toList(),
            ),
            const SizedBox(height: 20),
            Row(
              children: [
                Expanded(
                  child: _field(
                    'Сумма (₽)',
                    _amountCtrl,
                    keyboard: const TextInputType.numberWithOptions(
                      decimal: true,
                    ),
                    required: true,
                  ),
                ),
                const SizedBox(width: 8),
                IconButton(
                  icon: const Icon(Icons.camera_alt),
                  tooltip: 'Сканировать чек',
                  onPressed: _isScanning ? null : _scanReceipt,
                ),
              ],
            ),
            const SizedBox(height: 12),
            _field(
              'Пробег (км)',
              _mileageCtrl,
              keyboard: TextInputType.number,
              required: true,
            ),
            const SizedBox(height: 12),
            ListTile(
              contentPadding: EdgeInsets.zero,
              title: const Text('Дата'),
              subtitle: Text(
                '${_date.day.toString().padLeft(2, '0')}.${_date.month.toString().padLeft(2, '0')}.${_date.year}',
              ),
              trailing: const Icon(Icons.calendar_today),
              onTap: _pickDate,
            ),
            if (_type == ExpenseType.fuel) ...[
              const SizedBox(height: 12),
              Row(
                children: [
                  Expanded(
                    child: _field(
                      'Литры',
                      _litersCtrl,
                      keyboard: const TextInputType.numberWithOptions(
                        decimal: true,
                      ),
                    ),
                  ),
                  const SizedBox(width: 8),
                  IconButton(
                    icon: const Icon(
                      Icons.check_circle,
                      color: Colors.green,
                      size: 32,
                    ),
                    tooltip: 'Посчитать цену за литр',
                    onPressed: _calculatePrice,
                  ),
                ],
              ),
              const SizedBox(height: 12),
              Row(
                children: [
                  Expanded(
                    child: TextFormField(
                      controller: _pricePerLiterCtrl,
                      readOnly: true,
                      decoration: InputDecoration(
                        labelText: 'Цена за литр (₽)',
                        filled: true,
                        fillColor: Colors.grey.shade200,
                      ),
                      keyboardType: const TextInputType.numberWithOptions(
                        decimal: true,
                      ),
                    ),
                  ),
                  const SizedBox(width: 8),
                  IconButton(
                    icon: const Icon(
                      Icons.trending_up,
                      color: AppTheme.primary,
                    ),
                    tooltip: 'Динамика цен',
                    onPressed: () {
                      context.push('/stats');
                    },
                  ),
                ],
              ),
            ],
            if (_type != ExpenseType.fuel) ...[
              const SizedBox(height: 12),
              DropdownButtonFormField<String>(
                value: _isCustomService ? 'custom' : _selectedServiceId,
                decoration: const InputDecoration(labelText: 'Наименование'),
                items: [
                  ...ServiceItem.defaults.map((item) {
                    return DropdownMenuItem(
                      value: item.id,
                      child: Text('${item.name} (${item.category})'),
                    );
                  }),
                  const DropdownMenuItem(

  Widget _summary(double total, double fuelTotal, List<Expense> expenses) {
    final fmt = NumberFormat.currency(locale: 'ru_RU', symbol: '₽');
    double lPer100 = 0;

    final fuelExpenses = expenses
        .where((e) => e.type == ExpenseType.fuel)
        .toList();
    if (fuelExpenses.length >= 2) {
      fuelExpenses.sort((a, b) => a.mileage.compareTo(b.mileage));
      final first = fuelExpenses.first;
      final last = fuelExpenses.last;
      final distance = last.mileage - first.mileage;
      if (distance > 0) {
        final totalLiters = fuelExpenses.fold<double>(
          0,
          (sum, e) => sum + (e.liters ?? 0),
        );
        lPer100 = (totalLiters / distance) * 100;
      }
    }

    return Container(
      padding: const EdgeInsets.all(16),
      child: Row(
        children: [
          _statChip('Всего', fmt.format(total), Icons.payments),
          const SizedBox(width: 8),
          _statChip('Топливо', fmt.format(fuelTotal), Icons.local_gas_station),
          const SizedBox(width: 8),
          _statChip(
            'Расход',
            lPer100 > 0 ? '${lPer100.toStringAsFixed(1)} л/100' : '-',
            Icons.speed,
          ),
        ],
      ),
    );
  }

  Widget _upcomingServices(DataProvider data, Car car) {
    final expenses = data.expensesFor(widget.carId);
    final upcoming = <Widget>[];

    for (final item in ServiceItem.defaults) {
      Expense? last;
      for (final e in expenses) {
        if (e.serviceName == item.id) {
          last = e;
          break;
        }
      }

      if (last != null && item.defaultIntervalKm != null) {
        final nextMileage = last.mileage + item.defaultIntervalKm!.toInt();
        final remaining = nextMileage - car.mileage;

        if (remaining > 0 && remaining < 5000) {
          final isUrgent = remaining < 1000;
          upcoming.add(
            Container(
              padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 8),
              margin: const EdgeInsets.only(right: 8),
              decoration: BoxDecoration(
                color: isUrgent
                    ? Colors.red.withValues(alpha: 0.1)
                    : Colors.orange.withValues(alpha: 0.1),
                borderRadius: BorderRadius.circular(8),
              ),
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  Text(
                    item.name,
                    style: const TextStyle(
                      fontSize: 12,
                      fontWeight: FontWeight.bold,
                    ),
                  ),
                  Text(
                    'через $remaining км',
                    style: TextStyle(
                      fontSize: 11,
                      color: isUrgent ? Colors.red : Colors.orange,
                    ),
                  ),
                ],
              ),
            ),
          );
        }
      }
    }

    if (upcoming.isEmpty) return const SizedBox.shrink();
    return Container(
      height: 70,
      padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 8),
      child: ListView(scrollDirection: Axis.horizontal, children: upcoming),
    );
  }

  Widget _filterChips() {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Padding(
          padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 8),
          child: Row(
            children: [
              Expanded(
                child: OutlinedButton.icon(
                  icon: const Icon(Icons.calendar_today, size: 18),
                  label: Text(
                    _startDate == null && _endDate == null
                        ? 'Период: Все время'
                        : '${_startDate != null ? DateFormat('dd.MM').format(_startDate!) : '...'} - ${_endDate != null ? DateFormat('dd.MM').format(_endDate!) : '...'}',
                    overflow: TextOverflow.ellipsis,
                  ),
                  onPressed: _pickDateRange,
                ),
              ),
            ],
          ),
        ),
        Padding(
          padding: const EdgeInsets.symmetric(horizontal: 16),
          child: Wrap(
            spacing: 8,
            runSpacing: 8,
            children: [
              FilterChip(
                label: const Text('Все'),
                selected: _selectedFilter == null,
                onSelected: (_) => setState(() => _selectedFilter = null),
              ),
              ...ExpenseType.values.map((type) {
                return FilterChip(
                  label: Text('${type.icon} ${type.label}'),
                  selected: _selectedFilter == type,
                  onSelected: (_) => setState(() {
                    _selectedFilter = _selectedFilter == type ? null : type;
                  }),
                );
              }),
            ],
          ),
        ),
      ],
    );
  }

  Future<void> _pickDateRange() async {
    final now = DateTime.now();
    final picked = await showDateRangePicker(
      context: context,
      firstDate: DateTime(2000),
      lastDate: now,
      initialDateRange: _startDate != null && _endDate != null
          ? DateTimeRange(start: _startDate!, end: _endDate!)
          : null,
    );
    if (picked != null) {
      setState(() {
        _startDate = picked.start;
        _endDate = picked.end;
      });
    }
  }

  Widget _statChip(String label, String value, IconData icon) {
    return Expanded(
      child: Container(
        padding: const EdgeInsets.symmetric(vertical: 12, horizontal: 8),
        decoration: BoxDecoration(
          color: AppTheme.primary.withValues(alpha: 0.08),
          borderRadius: BorderRadius.circular(12),
        ),
        child: Column(
          children: [
            Icon(icon, color: AppTheme.primary, size: 20),
            const SizedBox(height: 4),
            Text(
              value,
              style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 14),
              overflow: TextOverflow.ellipsis,
            ),
            Text(
              label,
              style: const TextStyle(fontSize: 11, color: AppTheme.textMuted),
            ),
          ],
        ),
      ),
    );
  }

  Widget build(BuildContext context) {
    final data = context.watch<DataProvider>();
    final allFuel =
        data.cars
            .expand((c) => data.expensesFor(c.id))
            .where(
              (e) =>
                  e.type == ExpenseType.fuel &&
                  e.liters != null &&
                  e.liters! > 0,
            )
            .toList()
          ..sort((a, b) => a.date.compareTo(b.date));

    final filtered = allFuel.where((e) {
      if (_startDate != null && e.date.isBefore(_startDate!)) return false;
      if (_endDate != null &&
          e.date.isAfter(_endDate!.add(const Duration(days: 1))))
        return false;
      return true;
    }).toList();

    final fmt = NumberFormat.currency(locale: 'ru_RU', symbol: '₽');

    return Scaffold(
      appBar: AppBar(title: const Text('Динамика цен')),
      body: Column(
        children: [
          _dateFilterButton(),
          if (filtered.isEmpty)
            const Expanded(
              child: Center(
                child: Text(
                  'Нет данных за выбранный период',
                  style: TextStyle(color: AppTheme.textMuted),
                ),
              ),
            )
          else
            Expanded(
              child: ListView(
                padding: const EdgeInsets.all(16),
                children: [
                  _statsRow(filtered, fmt),
                  const SizedBox(height: 16),

  Widget _dateFilterButton() {
    return Padding(
      padding: const EdgeInsets.all(16),
      child: OutlinedButton.icon(
        icon: const Icon(Icons.date_range),
        label: Text(
          _startDate == null && _endDate == null
              ? 'Период: всё время'
              : '${DateFormat('dd.MM.yyyy').format(_startDate!)} — ${DateFormat('dd.MM.yyyy').format(_endDate!)}',
        ),
        onPressed: _pickDateRange,
      ),
    );
  }

  Future<void> _pickDateRange() async {
    final picked = await showDateRangePicker(
      context: context,
      firstDate: DateTime(2000),
      lastDate: DateTime.now(),
      initialDateRange: _startDate != null && _endDate != null
          ? DateTimeRange(start: _startDate!, end: _endDate!)
          : null,
    );
    if (picked != null) {
      setState(() {
        _startDate = picked.start;
        _endDate = picked.end;
      });
    }

  Widget _statsRow(List<Expense> expenses, fmt) {
    final prices = expenses
        .map((e) => e.pricePerLiter ?? (e.amount / e.liters!))
        .toList();
    final min = prices.reduce((a, b) => a < b ? a : b);
    final max = prices.reduce((a, b) => a > b ? a : b);
    final avg = prices.reduce((a, b) => a + b) / prices.length;

    return Row(
      children: [
        _statCard('Мин', fmt.format(min), Colors.green),
        const SizedBox(width: 8),
        _statCard('Сред', fmt.format(avg), AppTheme.primary),
        const SizedBox(width: 8),
        _statCard('Макс', fmt.format(max), Colors.red),
      ],
    );
  }

  Widget _statCard(String label, String value, Color color) {
    return Expanded(
      child: Container(
        padding: const EdgeInsets.symmetric(vertical: 12),
        decoration: BoxDecoration(
          color: color.withValues(alpha: 0.1),
          borderRadius: BorderRadius.circular(12),
        ),
        child: Column(
          children: [
            Text(label, style: TextStyle(fontSize: 12, color: color)),
            const SizedBox(height: 4),
            Text(
              value,
              style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 14),
            ),
          ],
        ),
      ),
    );
  }

  Widget _lineChart(List<Expense> expenses) {
    final spots = <FlSpot>[];
    for (int i = 0; i < expenses.length; i++) {
      final e = expenses[i];
      final price = e.pricePerLiter ?? (e.amount / e.liters!);
      spots.add(FlSpot(i.toDouble(), price));
    }

    final minY = spots.map((s) => s.y).reduce((a, b) => a < b ? a : b);
    final maxY = spots.map((s) => s.y).reduce((a, b) => a > b ? a : b);
    final padding = (maxY - minY) * 0.2;

    return SizedBox(
      height: 250,
      child: LineChart(
        LineChartData(
          minY: minY - padding,
          maxY: maxY + padding,
          lineTouchData: LineTouchData(
            touchTooltipData: LineTouchTooltipData(
              getTooltipItems: (spots) {
                return spots.map((spot) {
                  final e = expenses[spot.x.toInt()];
                  return LineTooltipItem(
                    '${DateFormat('dd.MM.yyyy').format(e.date)}\n${spot.y.toStringAsFixed(2)} ₽/л',
                    const TextStyle(color: Colors.white, fontSize: 12),
                  );
                }).toList();
              },
            ),
          ),
          lineBarsData: [
            LineChartBarData(
              spots: spots,
              isCurved: true,
              color: AppTheme.accent,
              barWidth: 3,
              dotData: const FlDotData(show: true),
              belowBarData: BarAreaData(
                show: true,
                color: AppTheme.accent.withValues(alpha: 0.1),
              ),
            ),
          ],
          titlesData: FlTitlesData(
            rightTitles: const AxisTitles(
              sideTitles: SideTitles(showTitles: false),
            ),
            topTitles: const AxisTitles(
              sideTitles: SideTitles(showTitles: false),
            ),
            leftTitles: AxisTitles(
              sideTitles: SideTitles(
                showTitles: true,
                reservedSize: 40,
                getTitlesWidget: (value, meta) {
                  return Text(
                    '${value.toStringAsFixed(0)} ₽',
                    style: const TextStyle(
                      fontSize: 10,
                      color: AppTheme.textMuted,
                    ),
                  );
                },
              ),
            ),
            bottomTitles: AxisTitles(
              sideTitles: SideTitles(
                showTitles: true,
                reservedSize: 30,
                interval: (expenses.length / 4).ceilToDouble(),
                getTitlesWidget: (value, meta) {
                  final index = value.toInt();
                  if (index < 0 || index >= expenses.length)
                    return const SizedBox.shrink();
                  return Padding(
                    padding: const EdgeInsets.only(top: 4),
                    child: Text(
                      DateFormat('dd.MM.yy').format(expenses[index].date),
                      style: const TextStyle(
                        fontSize: 9,
                        color: AppTheme.textMuted,
                      ),
                    ),
                  );
                },
              ),
            ),
          ),
          gridData: const FlGridData(show: true, drawVerticalLine: false),
          borderData: FlBorderData(show: false),
        ),
      ),
    );
  }

  Widget _priceList(List<Expense> expenses, fmt) {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        const Padding(
          padding: EdgeInsets.only(bottom: 8),
          child: Text(
            'История заправок',
            style: TextStyle(fontSize: 16, fontWeight: FontWeight.bold),
          ),
        ),
        ...expenses.reversed.map((e) {
          final price = e.pricePerLiter ?? (e.amount / e.liters!);

            margin: const EdgeInsets.only(bottom: 6),
            child: ListTile(
              contentPadding: const EdgeInsets.symmetric(
                horizontal: 12,
                vertical: 4,
              ),
              leading: const Text('⛽', style: TextStyle(fontSize: 20)),
              title: Text(
                '${e.liters!.toStringAsFixed(1)} л × ${price.toStringAsFixed(2)} ₽',
              ),
              subtitle: Text(
                DateFormat('dd.MM.yyyy').format(e.date),
                style: const TextStyle(fontSize: 12, color: AppTheme.textMuted),
              ),
              trailing: Text(
                fmt.format(e.amount),
                style: const TextStyle(fontWeight: FontWeight.bold),
              ),
            ),
          );
        }),
      ],
    );
  }

class AuthService {
  final FirebaseAuth _auth = FirebaseAuth.instance;

  User? get currentUser => _auth.currentUser;
  Stream<User?> get authStateChanges => _auth.authStateChanges();

  Future<User?> signInWithGoogle() async {
    if (kIsWeb) {
      // Web: используем GoogleAuthProvider
      final GoogleAuthProvider provider = GoogleAuthProvider();
      final UserCredential cred = await _auth.signInWithPopup(provider);
      return cred.user;
    } else {
      // Android: используем google_sign_in
      final GoogleSignInAccount? googleUser = await GoogleSignIn().signIn();
      if (googleUser == null) return null;

      final GoogleSignInAuthentication googleAuth =
          await googleUser.authentication;

      final credential = GoogleAuthProvider.credential(
        accessToken: googleAuth.accessToken,
        idToken: googleAuth.idToken,
      );

      final UserCredential cred = await _auth.signInWithCredential(credential);
      return cred.user;
    }
  }

  Future<void> signOut() async {
    await GoogleSignIn().signOut();
    await _auth.signOut();
  }
}

class DbService {

  Stream<List<Car>> watchCars(String userId) {
    return _db

        .snapshots()
        .map(
          (snap) => snap.docs.map((d) => Car.fromMap(d.data(), d.id)).toList(),
        );
  }

    final ref = await _db.collection('cars').add(car.toMap());
    return ref.id;
  }

    await _db.collection('cars').doc(car.id).update(car.toMap());
  }

  Future<void> deleteCar(String carId) async {
    await _db.collection('cars').doc(carId).delete();
    // Удаляем все расходы этой машины
    final expenses = await _db
        .collection('expenses')
        .where('carId', isEqualTo: carId)
        .get();
    for (final doc in expenses.docs) {
      await doc.reference.delete();
    }
  }

import 'package:image_picker/image_picker.dart';
import 'package:google_mlkit_text_recognition/google_mlkit_text_recognition.dart';

class OcrService {
  final ImagePicker _picker = ImagePicker();
  final TextRecognizer _recognizer = TextRecognizer();

  Future<Map<String, dynamic>?> scanReceipt() async {
    final XFile? image = await _picker.pickImage(source: ImageSource.camera);
    if (image == null) return null;

    final inputImage = InputImage.fromFile(File(image.path));
    final RecognizedText recognizedText = await _recognizer.processImage(
      inputImage,
    );

    String text = '';
    for (TextBlock block in recognizedText.blocks) {
      for (TextLine line in block.lines) {
        text += line.text + '\n';
      }
    }

    double? totalAmount;
    double? liters;
    DateTime? receiptDate;

    // Ищем сумму (ИТОГ, Итого, TOTAL)
    final totalRegex = RegExp(
      r'(?:ИТОГ|Итого|TOTAL|ИТОГО)[^\d]*(\d+[.,]\d+)',
      caseSensitive: false,
    );
    final totalMatch = totalRegex.firstMatch(text);
    if (totalMatch != null) {
      totalAmount = double.tryParse(totalMatch.group(1)!.replaceAll(',', '.'));
    }

    // Ищем литраж (число перед "л", "литр", "л.")
    final litersRegex = RegExp(
      r'(\d+[.,]\d+)\s*(?:л|литр|л\.)',
      caseSensitive: false,
    );
    final litersMatch = litersRegex.firstMatch(text);
    if (litersMatch != null) {
      liters = double.tryParse(litersMatch.group(1)!.replaceAll(',', '.'));
    }

    // Ищем дату в формате ДД.ММ.ГГГГ или ДД/ММ/ГГГГ
    final dateRegex = RegExp(r'(\d{2})[./](\d{2})[./](\d{4})');
    final dateMatch = dateRegex.firstMatch(text);
    if (dateMatch != null) {
      final day = int.tryParse(dateMatch.group(1)!);
      final month = int.tryParse(dateMatch.group(2)!);
      final year = int.tryParse(dateMatch.group(3)!);
      if (day != null && month != null && year != null) {
        try {
          receiptDate = DateTime(year, month, day);
        } catch (_) {}
      }
    }

    // Если сумму не нашли по ключевым словам, берем самое большое число в чеке
    if (totalAmount == null) {
      final numbers = RegExp(r'\d+[.,]\d+').allMatches(text);
      double maxNum = 0;
      for (var match in numbers) {
        final num = double.tryParse(match.group(0)!.replaceAll(',', '.')) ?? 0;
        if (num > maxNum) maxNum = num;
      }
      if (maxNum > 0) totalAmount = maxNum;
    }

    return {'amount': totalAmount, 'liters': liters, 'date': receiptDate};
  }

  void dispose() {
    _recognizer.close();
  }
}

class UpdateService {
  // ЗАМЕНИ 'YOUR_GITHUB_USERNAME' на свой логин GitHub, а 'autodiary' на имя репозитория
  static const String repoOwner = 'ak3301389';
  static const String repoName = 'autodiary';

  static Future<Map<String, dynamic>?> checkForUpdate() async {
    try {
      final packageInfo = await PackageInfo.fromPlatform();
      final currentVersion = packageInfo.version; // например, "1.0.0"

      final response = await http.get(
        Uri.parse(
          'https://api.github.com/repos/$repoOwner/$repoName/releases/latest',
        ),
        headers: {'Accept': 'application/vnd.github.v3+json'},
      );

      if (response.statusCode == 200) {
        final data = json.decode(response.body);
        final latestVersionTag =
            data['tag_name'] as String; // например, "v1.0.1"
        final cleanLatestVersion = latestVersionTag.startsWith('v')
            ? latestVersionTag.substring(1)
            : latestVersionTag;

        if (_isNewerVersion(cleanLatestVersion, currentVersion)) {
          return {
            'hasUpdate': true,
            'latestVersion': cleanLatestVersion,
            'releaseNotes': data['body'] ?? 'Исправления и улучшения',
            'downloadUrl': data['html_url'], // Ссылка на релиз на GitHub
          };
        }
      }
      return {'hasUpdate': false};
    } catch (e) {
      print('Ошибка проверки обновлений: $e');
      return {'hasUpdate': false};
    }
  }

  static bool _isNewerVersion(String latest, String current) {
    final latestParts = latest.split('.').map(int.parse).toList();
    final currentParts = current.split('.').map(int.parse).toList();

    for (int i = 0; i < 3; i++) {
      final l = latestParts.length > i ? latestParts[i] : 0;
      final c = currentParts.length > i ? currentParts[i] : 0;
      if (l > c) return true;
      if (l < c) return false;
    }
    return false;
  }

  static Future<void> openDownloadPage(String url) async {
    final uri = Uri.parse(url);
    if (await canLaunchUrl(uri)) {
